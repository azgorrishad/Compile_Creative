   271→   271→function TypewriterText({ text, delay = 0 }: { text: string; delay?: number }) {
   272→   272→  const [displayText, setDisplayText] = useState("");
   273→   273→  const [isTyping, setIsTyping] = useState(false);
   274→   274→  const mounted = useMounted();
   275→   275→  const ref = useRef(null);
   276→   276→  const isInView = useInView(ref, { once: true });
   277→   277→
   278→   278→  useEffect(() => {
   279→   279→    if (!isInView || !mounted) return;
   280→   280→    let i = 0;
   281→   281→    const timeout = setTimeout(() => {
   282→   282→      setIsTyping(true);
   283→   283→      const interval = setInterval(() => {
   284→   284→        if (i < text.length) {
   285→   285→          setDisplayText(text.slice(0, i + 1));
   286→   286→          i++;
   287→   287→        } else {
   288→   288→          setIsTyping(false);
   289→   289→          clearInterval(interval);
   290→   290→        }
   291→   291→      }, 40);
   292→   292→      return () => clearInterval(interval);
   293→   293→    }, delay);
   294→   294→    return () => clearTimeout(timeout);
   295→   295→  }, [isInView, text, delay, mounted]);
   296→   296→
   297→   297→  if (!mounted) {
   298→   298→    return <span ref={ref}>{text}</span>;
   299→   299→  }
   300→   300→
   301→   301→  return (
   302→   302→    <span ref={ref} className={isTyping ? "typewriter-cursor" : ""} suppressHydrationWarning>
   303→   303→      {displayText || text}
   304→   304→    </span>
   305→   305→  );
   306→   306→}

   310→   310→function RotatingWords({ words, interval = 2500 }: { words: string[]; interval?: number }) {
   311→   311→  const [index, setIndex] = useState(0);

   320→   320→        setIndex((prev) => (prev + 1) % words.length);

   342→   342→/* ─── 3D Tilt Card Component ──────────────────────── */
   343→   343→
   344→   344→function TiltCard({ children, className = "" }: { children: React.ReactNode; className?: string }) {
   345→   345→  const cardRef = useRef<HTMLDivElement>(null);
   346→   346→
   347→   347→  const handleMouseMove = useCallback((e: React.MouseEvent<HTMLDivElement>) => {
   348→   348→    const card = cardRef.current;
   349→   349→    if (!card) return;
   350→   350→    const rect = card.getBoundingClientRect();
   351→   351→    const x = e.clientX - rect.left;
   352→   352→    const y = e.clientY - rect.top;
   353→   353→    const centerX = rect.width / 2;
   354→   354→    const centerY = rect.height / 2;
   355→   355→    const rotateX = (y - centerY) / centerY * -5;
   356→   356→    const rotateY = (x - centerX) / centerX * 5;
   357→   357→    card.style.transform = `perspective(1200px) rotateX(${rotateX}deg) rotateY(${rotateY}deg) scale3d(1.01, 1.01, 1.01)`;
   358→   358→    card.style.setProperty("--mouse-x", `${x}px`);
   359→   359→    card.style.setProperty("--mouse-y", `${y}px`);
   360→   360→  }, []);
   361→   361→
   362→   362→  const handleMouseLeave = useCallback(() => {
   363→   363→    const card = cardRef.current;
   364→   364→    if (!card) return;
   365→   365→    card.style.transform = "perspective(1200px) rotateX(0) rotateY(0) scale3d(1, 1, 1)";
   366→   366→  }, []);
   367→   367→
   368→   368→  return (
   369→   369→    <div
   370→   370→      ref={cardRef}
   371→   371→      className={`tilt-card ${className}`}
   372→   372→      onMouseMove={handleMouseMove}
   373→   373→      onMouseLeave={handleMouseLeave}
   374→   374→    >
   375→   375→      <div className="tilt-card-shine" />
   376→   376→      {children}
   377→   377→    </div>
   378→   378→  );
   379→   379→}

   381→   381→/* ─── Animated Counter Component ────────────────── */
   382→   382→
   383→   383→function AnimatedCounter({ value, suffix = "" }: { value: string; suffix?: string }) {
   384→   384→  const ref = useRef(null);
   385→   385→  const isInView = useInView(ref, { once: true });
   386→   386→  const [count, setCount] = useState(0);
   387→   387→  const mounted = useMounted();
   388→   388→  const numericValue = parseInt(value.replace(/[^0-9]/g, ""));
   389→   389→  const prefix = value.match(/^[^0-9]*/)?.[0] || "";
   390→   390→  const suffixMatch = value.match(/[^0-9]*$/)?.[0] || "";
   391→   391→
   392→   392→  useEffect(() => {
   393→   393→    if (!isInView || isNaN(numericValue) || !mounted) return;
   394→   394→    const duration = 1500;
   395→   395→    const startTime = Date.now();
   396→   396→    const animate = () => {
   397→   397→      const elapsed = Date.now() - startTime;
   398→   398→      const progress = Math.min(elapsed / duration, 1);
   399→   399→      const eased = 1 - Math.pow(1 - progress, 3);
   400→   400→      setCount(Math.round(numericValue * eased));
   401→   401→      if (progress < 1) requestAnimationFrame(animate);
   402→   402→    };
   403→   403→    requestAnimationFrame(animate);
   404→   404→  }, [isInView, numericValue, mounted]);
   405→   405→
   406→   406→  if (!mounted) {
   407→   407→    return (
   408→   408→      <span ref={ref} className="stat-number" suppressHydrationWarning>
   409→   409→        {prefix}{value}{suffixMatch}{suffix}
   410→   410→      </span>
   411→   411→    );
   412→   412→  }
   413→   413→
   414→   414→  if (isNaN(numericValue)) {
   415→   415→    return (
   416→   416→      <motion.span
   417→   417→        ref={ref}
   418→   418→        initial={{ opacity: 0, y: 20 }}
   419→   419→        animate={isInView ? { opacity: 1, y: 0 } : {}}
   420→   420→        transition={{ duration: 0.6, ease: agencyEase }}
   421→   421→        className="stat-number"
   422→   422→        suppressHydrationWarning
   423→   423→      >
   424→   424→        {isInView ? value : value}
   425→   425→        {suffix}
   426→   426→      </motion.span>
   427→   427→    );
   428→   428→  }
   429→   429→
   430→   430→  return (
   431→   431→    <motion.span
   432→   432→      ref={ref}
   433→   433→      initial={{ opacity: 0, y: 20 }}
   434→   434→      animate={isInView ? { opacity: 1, y: 0 } : {}}
   435→   435→      transition={{ duration: 0.6, ease: agencyEase }}
   436→   436→      className="stat-number"
   437→   437→      suppressHydrationWarning
   438→   438→    >
   439→   439→      {isInView ? `${prefix}${count}${suffixMatch}` : `${prefix}0${suffixMatch}`}
   440→   440→      {suffix}
   441→   441→    </motion.span>
   442→   442→  );
   443→   443→}
