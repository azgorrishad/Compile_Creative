   271→function TypewriterText({ text, delay = 0 }: { text: string; delay?: number }) {
   272→  const [displayText, setDisplayText] = useState("");
   273→  const [isTyping, setIsTyping] = useState(false);
   274→  const mounted = useMounted();
   275→  const ref = useRef(null);
   276→  const isInView = useInView(ref, { once: true });
   277→
   278→  useEffect(() => {
   279→    if (!isInView || !mounted) return;
   280→    let i = 0;
   281→    const timeout = setTimeout(() => {
   282→      setIsTyping(true);
   283→      const interval = setInterval(() => {
   284→        if (i < text.length) {
   285→          setDisplayText(text.slice(0, i + 1));
   286→          i++;
   287→        } else {
   288→          setIsTyping(false);
   289→          clearInterval(interval);
   290→        }
   291→      }, 40);
   292→      return () => clearInterval(interval);
   293→    }, delay);
   294→    return () => clearTimeout(timeout);
   295→  }, [isInView, text, delay, mounted]);
   296→
   297→  if (!mounted) {
   298→    return <span ref={ref}>{text}</span>;
   299→  }
   300→
   301→  return (
   302→    <span ref={ref} className={isTyping ? "typewriter-cursor" : ""} suppressHydrationWarning>
   303→      {displayText || text}
   304→    </span>
   305→  );
   306→}

   308→/* ─── Rotating Words Component ────────────────────── */
   309→
   310→function RotatingWords({ words, interval = 2500 }: { words: string[]; interval?: number }) {
   311→  const [index, setIndex] = useState(0);
   312→  const [isAnimating, setIsAnimating] = useState(true);
   313→  const mounted = useMounted();
   314→
   315→  useEffect(() => {
   316→    if (!mounted) return;
   317→    const timer = setInterval(() => {
   318→      setIsAnimating(false);
   319→      setTimeout(() => {
   320→        setIndex((prev) => (prev + 1) % words.length);
   321→        setIsAnimating(true);
   322→      }, 400);
   323→    }, interval);
   324→    return () => clearInterval(timer);
   325→  }, [words.length, interval, mounted]);
   326→
   327→  if (!mounted) {
   328→    return <span suppressHydrationWarning>{words[0]}</span>;
   329→  }
   330→
   331→  return (
   332→    <span className="inline-block overflow-hidden" style={{ perspective: "600px" }} suppressHydrationWarning>
   333→      <span
   334→        className={`inline-block ${isAnimating ? "animate-word-in" : "animate-word-out"}`}
   335→      >
   336→        {words[index]}
   337→      </span>
   338→    </span>
   339→  );
   340→}

   342→/* ─── 3D Tilt Card Component ──────────────────────── */
   343→
   344→function TiltCard({ children, className = "" }: { children: React.ReactNode; className?: string }) {
   345→  const cardRef = useRef<HTMLDivElement>(null);
   346→
   347→  const handleMouseMove = useCallback((e: React.MouseEvent<HTMLDivElement>) => {
   348→    const card = cardRef.current;
   349→    if (!card) return;
   350→    const rect = card.getBoundingClientRect();
   351→    const x = e.clientX - rect.left;
   352→    const y = e.clientY - rect.top;
   353→    const centerX = rect.width / 2;
   354→    const centerY = rect.height / 2;
   355→    const rotateX = (y - centerY) / centerY * -5;
   356→    const rotateY = (x - centerX) / centerX * 5;
   357→    card.style.transform = `perspective(1200px) rotateX(${rotateX}deg) rotateY(${rotateY}deg) scale3d(1.01, 1.01, 1.01)`;
   358→    card.style.setProperty("--mouse-x", `${x}px`);
   359→    card.style.setProperty("--mouse-y", `${y}px`);
   360→  }, []);
   361→
   362→  const handleMouseLeave = useCallback(() => {
   363→    const card = cardRef.current;
   364→    if (!card) return;
   365→    card.style.transform = "perspective(1200px) rotateX(0) rotateY(0) scale3d(1, 1, 1)";
   366→  }, []);
   367→
   368→  return (
   369→    <div
   370→      ref={cardRef}
   371→      className={`tilt-card ${className}`}
   372→      onMouseMove={handleMouseMove}
   373→      onMouseLeave={handleMouseLeave}
   374→    >
   375→      <div className="tilt-card-shine" />
   376→      {children}
   377→    </div>
   378→  );
   379→}

   381→/* ─── Animated Counter Component ────────────────── */
   382→
   383→function AnimatedCounter({ value, suffix = "" }: { value: string; suffix?: string }) {
   384→  const ref = useRef(null);
   385→  const isInView = useInView(ref, { once: true });
   386→  const [count, setCount] = useState(0);
   387→  const mounted = useMounted();
   388→  const numericValue = parseInt(value.replace(/[^0-9]/g, ""));
   389→  const prefix = value.match(/^[^0-9]*/)?.[0] || "";
   390→  const suffixMatch = value.match(/[^0-9]*$/)?.[0] || "";
   391→
   392→  useEffect(() => {
   393→    if (!isInView || isNaN(numericValue) || !mounted) return;
   394→    const duration = 1500;
   395→    const startTime = Date.now();
   396→    const animate = () => {
   397→      const elapsed = Date.now() - startTime;
   398→      const progress = Math.min(elapsed / duration, 1);
   399→      const eased = 1 - Math.pow(1 - progress, 3);
   400→      setCount(Math.round(numericValue * eased));
   401→      if (progress < 1) requestAnimationFrame(animate);
   402→    };
   403→    requestAnimationFrame(animate);
   404→  }, [isInView, numericValue, mounted]);
   405→
   406→  if (!mounted) {
   407→    return (
   408→      <span ref={ref} className="stat-number" suppressHydrationWarning>
   409→        {prefix}{value}{suffixMatch}{suffix}
   410→      </span>
   411→    );
   412→  }
   413→
   414→  if (isNaN(numericValue)) {
   415→    return (
   416→      <motion.span
   417→        ref={ref}
   418→        initial={{ opacity: 0, y: 20 }}
   419→        animate={isInView ? { opacity: 1, y: 0 } : {}}
   420→        transition={{ duration: 0.6, ease: agencyEase }}
   421→        className="stat-number"
   422→        suppressHydrationWarning
   423→      >
   424→        {isInView ? value : value}
   425→        {suffix}
   426→      </motion.span>
   427→    );
   428→  }
   429→
   430→  return (
   431→    <motion.span
   432→      ref={ref}
   433→      initial={{ opacity: 0, y: 20 }}
   434→      animate={isInView ? { opacity: 1, y: 0 } : {}}
   435→      transition={{ duration: 0.6, ease: agencyEase }}
   436→      className="stat-number"
   437→      suppressHydrationWarning
   438→    >
   439→      {isInView ? `${prefix}${count}${suffixMatch}` : `${prefix}0${suffixMatch}`}
   440→      {suffix}
   441→    </motion.span>
   442→  );
   443→}

   917→
   918→function CostOfLookingAverage() {
   919→  const [currentSlide, setCurrentSlide] = useState(0);
   920→  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
   921→
   922→  const startInterval = useCallback(() => {
   923→    if (intervalRef.current) clearInterval(intervalRef.current);
   924→    intervalRef.current = setInterval(() => {
   925→      setCurrentSlide((prev) => (prev + 1) % CAROUSEL_SLIDES.length);
   926→    }, 4000);
   927→  }, []);
   928→
   929→  useEffect(() => {
   930→    startInterval();
   931→    return () => {
   932→      if (intervalRef.current) clearInterval(intervalRef.current);
   933→    };
   934→  }, [startInterval]);
   935→
   936→  const goToSlide = (index: number) => {
   937→    setCurrentSlide(index);
   938→    startInterval();
   939→  };
